cmake_minimum_required(VERSION 3.20)
project(InazumaElevenPC VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ie1_rom STATIC src/nds_rom.cpp)
target_include_directories(ie1_rom PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
add_executable(ie1_romtool src/main.cpp)
target_link_libraries(ie1_romtool PRIVATE ie1_rom)
if(MSVC)
  target_compile_options(ie1_rom PRIVATE /W4)
  target_compile_options(ie1_romtool PRIVATE /W4)
else()
  target_compile_options(ie1_rom PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(ie1_romtool PRIVATE -Wall -Wextra -Wpedantic)
endif()
